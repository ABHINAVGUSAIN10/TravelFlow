+function getDistance(lat1: number, lon1: number, lat2: number, lon2: number) {
+  const R = 6371;
+  const dLat = ((lat2 - lat1) * Math.PI) / 180;
+  const dLon = ((lon2 - lon1) * Math.PI) / 180;
+  const a = Math.sin(dLat/2)**2 + Math.cos(lat1*Math.PI/180)*Math.cos(lat2*Math.PI/180)*Math.sin(dLon/2)**2;
+  return R * 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a));

+    if (data.code !== "Ok" || !data.routes?.[0]) return null;
+    const rt = data.routes[0];
+    return {
+      path: rt.geometry.coordinates.map(([lg, la]: [number,number]) => [la, lg] as [number,number]),
+      km: rt.distance / 1000,
+      mins: rt.duration / 60,

+function flightArc(lat1: number, lon1: number, lat2: number, lon2: number, n=80): [number,number][] {
+  const rad = (d: number) => d*Math.PI/180, deg = (r: number) => r*180/Math.PI;
+  const ╧å1=rad(lat1), ╬╗1=rad(lon1), ╧å2=rad(lat2), ╬╗2=rad(lon2);
+  const d = 2*Math.asin(Math.sqrt(Math.sin((╧å2-╧å1)/2)**2+Math.cos(╧å1)*Math.cos(╧å2)*Math.sin((╬╗2-╬╗1)/2)**2));
+  if (d < 0.0001) return [[lat1,lon1],[lat2,lon2]];
+  return Array.from({length:n+1}, (_,i) => {
+    const t=i/n, A=Math.sin((1-t)*d)/Math.sin(d), B=Math.sin(t*d)/Math.sin(d);
+    const x=A*Math.cos(╧å1)*Math.cos(╬╗1)+B*Math.cos(╧å2)*Math.cos(╬╗2);
+    const y=A*Math.cos(╧å1)*Math.sin(╬╗1)+B*Math.cos(╧å2)*Math.sin(╬╗2);
+    const z=A*Math.sin(╧å1)+B*Math.sin(╧å2);
+    return [deg(Math.atan2(z,Math.sqrt(x*x+y*y))), deg(Math.atan2(y,x))] as [number,number];
+  });

+/**
+ * Geocode a place name using the Nominatim (OpenStreetMap) API.
+ * Completely free, no API key required.

+export interface GeoPoint {
+  lat: number;
+  lng: number;
+  displayName: string;
+}

+export async function geocodePlace(query: string): Promise<GeoPoint | null> {
+  try {
+    const url = `https://nominatim.openstreetmap.org/search?q=${encodeURIComponent(
+      query + ", India"
+    )}&format=json&limit=1`;
+
+    const headers: Record<string, string> = { "Accept-Language": "en" };
+    if (typeof window === "undefined") {

+    }
+
+    const res = await fetch(url, { headers });
+
+    if (!res.ok) return null;
+    const data = await res.json();
+
+    if (!data || data.length === 0) return null;
+
+    return {
+      lat: parseFloat(data[0].lat),
+      lng: parseFloat(data[0].lon),
+      displayName: data[0].display_name,
+    };
+  } catch {
+    return null;
+  }

+
+    if (!data || !data.address) return null;
+
+    // Pick the most specific available place name

+      addr.town ||
+      addr.village ||
+      addr.county ||
+      addr.state_district ||

+import mongoose from "mongoose";
+import { DESTINATIONS } from "../lib/data";
+
+const MONGODB_URI = process.env.MONGODB_URI || "mongodb://localhost:27017/travelflow";
+
+const LocationSchema = new mongoose.Schema({
+  title: String,
+  subtitle: String,
+  description: String,
+  bgImage: String,
+  cardImage: String,
+  gradient: String,
+  accentColor: String,
+  coordinates: String,
+  state: String,
+  category: String,
+  rating: Number,
+  reviews: Number,
+  priceLabel: String,
+  mustSee: [String],
+  bestTime: String,
+  idealDays: String,
+});
+
+const Location = mongoose.models.Location || mongoose.model("Location", LocationSchema);
+
+async function seedDatabase() {
+  try {
+    console.log("Connecting to MongoDB at", MONGODB_URI);
+    await mongoose.connect(MONGODB_URI);
+    
+    console.log("Connected to database. Emptying existing Locations collection...");
+    await Location.deleteMany({});
+    
+    console.log("Seeding with the following data:");
+    console.log(DESTINATIONS.map(l => l.title));
+    
+    for (const loc of DESTINATIONS) {
+      // Create valid DB objects omitting the original ID
+      const { id, ...data } = loc;
+      const doc = new Location(data);
+      await doc.save();
+    }
+    
+    console.log(`Successfully seeded ${DESTINATIONS.length} locations!`);
+  } catch (error) {
+    console.error("Error seeding database:", error);
+  } finally {
+    console.log("Closing connection...");
+    await mongoose.connection.close();
+    process.exit(0);
+  }
+}
+
+seedDatabase();
